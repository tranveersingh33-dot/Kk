function calculateJantri(text) { let total = 0; const results = []; const lines = text.split(/\r?\n/); lines.forEach(line => { // Number groups followed by a bracket value const regex = /([^()]*?)\s*\(\s*(\d+)\s*\)?/g; let match; while ((match = regex.exec(line)) !== null) { const before = match[1]; // Treat commas, dots, spaces and symbols as separators const numbers = before.match(/\d+/g) || []; if (numbers.length === 0) continue; const multiplier = Number(match[2]); const count = numbers.length; const result = count * multiplier; total += result; results.push( `${numbers.join(', ')} (${count} × ${multiplier} = ${result})` ); } }); results.push('--------------------'); results.push(`GRAND TOTAL = ${total}`); return results.join('\n'); } function calculateButton() { const input = document.getElementById('calcText').value; document.getElementById('calcOutput').value = calculateJantri(input); }
